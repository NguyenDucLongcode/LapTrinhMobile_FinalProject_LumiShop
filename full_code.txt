// api/axiosClient.ts
// Cấu hình Axios client để gọi Mock API (JSON Server)
import axios from 'axios';
import { Platform } from 'react-native';
import Constants from 'expo-constants';

// Lấy IP động từ Expo hostUri để hỗ trợ Android Emulator và điện thoại thật qua Expo Go
// - Web / iOS Simulator: dùng localhost
// - Android Emulator: Expo hostUri thường là 10.0.2.2 hoặc IP LAN
// - Điện thoại thật qua Expo Go: IP LAN của máy tính host
function getBaseUrl(): string {
  if (Platform.OS === 'web') {
    return 'http://localhost:3000';
  }
  // Lấy hostUri từ Expo (ví dụ: "192.168.1.5:8081")
  const hostUri = Constants.expoConfig?.hostUri;
  if (hostUri) {
    const host = hostUri.split(':')[0]; // Chỉ lấy phần IP, bỏ port 8081
    return `http://${host}:3000`;
  }
  // Fallback cho Android Emulator nếu không lấy được hostUri
  if (Platform.OS === 'android') {
    return 'http://10.0.2.2:3000';
  }
  return 'http://localhost:3000';
}

export const BASE_URL = getBaseUrl();

export const axiosClient = axios.create({
  baseURL: BASE_URL,
  headers: {
    'Content-Type': 'application/json',
    Accept: 'application/json',
  },
  timeout: 10000, // 10 giây timeout
});

// Request interceptor: có thể bổ sung token xác thực vào header nếu có
axiosClient.interceptors.request.use(
  (config) => {
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// Response interceptor: chuẩn hóa xử lý phản hồi hoặc lỗi
axiosClient.interceptors.response.use(
  (response) => {
    return response;
  },
  (error) => {
    const message =
      error.response?.data?.message ||
      error.message ||
      'Đã xảy ra lỗi khi kết nối tới máy chủ!';
    return Promise.reject(new Error(message));
  }
);

export default axiosClient;

// app/(app)/admin/products.tsx
// Quản lý Sản phẩm LumiShop: Danh sách sản phẩm bo góc 14, Nút Thêm mới bo tròn 24 và Modal nhập liệu hiện đại

import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  FlatList,
  StyleSheet,
  Pressable,
  Modal,
  ScrollView,
  SafeAreaView,
  Alert,
  Platform,
  ActivityIndicator,
} from 'react-native';
import { Image } from 'expo-image';
import { Redirect } from 'expo-router';
import { useQueryClient } from '@tanstack/react-query';
import { useProducts } from '../../../hooks/useProducts';
import { useTheme } from '../../../theme/useTheme';
import { Header } from '../../../components/Header';
import { formatCurrency } from '../../../utils/calculateTotal';
import axiosClient from '../../../api/axiosClient';
import { Product } from '../../../types';
import { useAuthStore } from '../../../store/authStore';

export default function AdminProductsScreen() {
  const { colors, borderRadius, shadow, shadowSm } = useTheme();
  const queryClient = useQueryClient();
  const { data: products, isLoading } = useProducts();
  const { user } = useAuthStore();

  // Tất cả useState phải khai báo trước mọi early return (Rules of Hooks)
  const [modalVisible, setModalVisible] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);

  // Form input
  const [name, setName] = useState('');
  const [price, setPrice] = useState('');
  const [category, setCategory] = useState('clothing');
  const [stock, setStock] = useState('20');
  const [image, setImage] = useState(
    'https://images.unsplash.com/photo-1523275335684-37898b6baf30?w=500&auto=format&fit=crop&q=80'
  );
  const [description, setDescription] = useState('');

  // Bảo vệ phân quyền: chỉ admin mới được vào (sau tất cả hooks)
  if (!user || user.role !== 'admin') {
    return <Redirect href="/(app)" />;
  }

  const showAlert = (title: string, message: string) => {
    if (Platform.OS === 'web') {
      alert(`${title}: ${message}`);
    } else {
      Alert.alert(title, message);
    }
  };

  const resetForm = () => {
    setName('');
    setPrice('');
    setCategory('clothing');
    setStock('20');
    setDescription('');
    setModalVisible(false);
  };

  const handleCreateProduct = async () => {
    if (!name.trim()) {
      showAlert('Lỗi', 'Vui lòng nhập tên sản phẩm!');
      return;
    }
    const numPrice = Number(price);
    if (!numPrice || numPrice <= 0) {
      showAlert('Lỗi', 'Vui lòng nhập giá hợp lệ (> 0)!');
      return;
    }
    // Validate stock: chấp nhận 0 (hết hàng), không cho số âm
    const numStock = parseInt(stock, 10);
    if (isNaN(numStock) || numStock < 0) {
      showAlert('Lỗi', 'Tồn kho phải là số nguyên không âm (0 hoặc lớn hơn)!');
      return;
    }

    setIsSubmitting(true);
    const newProduct: Product = {
      id: String(Date.now()),
      name: name.trim(),
      price: numPrice,
      category,
      stock: numStock,
      image: image.trim(),
      description: description.trim() || 'Sản phẩm mới thêm tại LumiShop',
      rating: 5,
    };

    try {
      await axiosClient.post('/products', newProduct);
      setIsSubmitting(false);
      resetForm();
      queryClient.invalidateQueries({ queryKey: ['products'] });
      showAlert('Thành công', 'Đã thêm sản phẩm mới vào LumiShop!');
    } catch (e: any) {
      setIsSubmitting(false);
      console.warn('Lỗi gọi API thêm sản phẩm:', e);
      const errMsg = e?.message || 'Không thể thêm sản phẩm. Kiểm tra JSON Server!';
      showAlert('Lỗi', `Không thể thêm sản phẩm!\n${errMsg}`);
    }
  };

  const handleDeleteProduct = (productId: string, productName: string) => {
    const doDelete = async () => {
      try {
        await axiosClient.delete(`/products/${productId}`);
        queryClient.invalidateQueries({ queryKey: ['products'] });
      } catch (e: any) {
        console.warn('Lỗi gọi API xóa sản phẩm:', e);
        showAlert('Lỗi xóa sản phẩm', e?.message || 'Không thể xóa sản phẩm do JSON Server offline!');
      }
    };

    if (Platform.OS === 'web') {
      if (window.confirm(`Bạn có chắc muốn xóa "${productName}"?`)) {
        doDelete();
      }
    } else {
      Alert.alert('Xác nhận xóa', `Bạn có chắc muốn xóa "${productName}"?`, [
        { text: 'Hủy', style: 'cancel' },
        { text: 'Xóa', style: 'destructive', onPress: doDelete },
      ]);
    }
  };

  return (
    <SafeAreaView style={[styles.container, { backgroundColor: colors.bg }]}>
      <Header
        title="Sản phẩm LumiShop"
        showBack={true}
        showCart={false}
        rightAction={
          <Pressable
            accessible={true}
            accessibilityRole="button"
            accessibilityLabel="Thêm sản phẩm"
            onPress={() => setModalVisible(true)}
            style={[
              styles.topAddBtn,
              {
                backgroundColor: colors.accent,
                borderRadius: borderRadius.pill,
              },
            ]}
          >
            <Text style={styles.topAddText}>+ Thêm mới</Text>
          </Pressable>
        }
      />

      {isLoading ? (
        <View style={styles.centerBox}>
          <ActivityIndicator size="large" color={colors.accent} />
          <Text style={[styles.loadingText, { color: colors.muted }]}>
            Đang tải danh sách sản phẩm...
          </Text>
        </View>
      ) : (
        <FlatList
          data={products}
          keyExtractor={(item) => String(item.id)}
          contentContainerStyle={styles.listContent}
          renderItem={({ item }) => (
            <View
              style={[
                styles.itemRow,
                {
                  backgroundColor: colors.card,
                  borderRadius: borderRadius.card,
                  borderColor: colors.border,
                },
                shadowSm,
              ]}
            >
              <Image
                source={{ uri: item.image }}
                style={[styles.thumbImage, { borderRadius: borderRadius.input }]}
                contentFit="cover"
                transition={200}
              />

              <View style={styles.itemDetails}>
                <Text numberOfLines={1} style={[styles.prodName, { color: colors.text }]}>
                  {item.name}
                </Text>
                <Text style={[styles.prodPrice, { color: colors.accent }]}>
                  {formatCurrency(item.price)}
                </Text>
                <Text style={[styles.prodMeta, { color: colors.muted }]}>
                  Kho: {item.stock} • Loại: {item.category} • ⭐ {item.rating ?? 5.0}
                </Text>
              </View>

              <Pressable
                accessible={true}
                accessibilityRole="button"
                accessibilityLabel={`Xóa ${item.name}`}
                onPress={() => handleDeleteProduct(item.id, item.name)}
                style={({ pressed }) => [
                  styles.deleteCircleBtn,
                  {
                    backgroundColor: '#FEE2E2',
                    transform: [{ scale: pressed ? 0.9 : 1 }],
                  },
                ]}
              >

import {
  View,
  Text,
  TextInput,
  FlatList,
  StyleSheet,
  Pressable,
  Modal,
  ScrollView,
  SafeAreaView,
  Alert,
  Platform,
  ActivityIndicator,
} from 'react-native';
import { Redirect } from 'expo-router';
import { useTheme } from '../../../theme/useTheme';
import { Header } from '../../../components/Header';
import { formatCurrency } from '../../../utils/calculateTotal';
import axiosClient from '../../../api/axiosClient';
import { initialVouchers } from '../../../data/products';
import { Voucher } from '../../../types';
import { useAuthStore } from '../../../store/authStore';

export default function AdminVouchersScreen() {
  const { colors, borderRadius, shadow, shadowSm } = useTheme();
  const { user } = useAuthStore();

  const [vouchers, setVouchers] = useState<Voucher[]>(initialVouchers);
  const [isLoading, setIsLoading] = useState(false);
  const [modalVisible, setModalVisible] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);

  // Form input
  const [code, setCode] = useState('');
  const [discount, setDiscount] = useState('');
  const [minOrder, setMinOrder] = useState('200000');

  const fetchVouchers = async () => {
    setIsLoading(true);
    try {
      const res = await axiosClient.get<Voucher[]>('/vouchers');
      if (res.data && Array.isArray(res.data) && res.data.length > 0) {
        setVouchers(res.data);
      } else {
        setVouchers(initialVouchers);
      }
    } catch (e) {
      console.warn('Lỗi fetch vouchers từ server, dùng initialVouchers:', e);
      setVouchers(initialVouchers);
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchVouchers();
  }, []);

  // Bảo vệ phân quyền: chỉ admin mới được vào
  if (!user || user.role !== 'admin') {
    return <Redirect href="/(app)" />;
  }

  const showAlert = (title: string, message: string) => {
    if (Platform.OS === 'web') {
      alert(`${title}: ${message}`);
    } else {
      Alert.alert(title, message);
    }
  };

  const resetForm = () => {
    setCode('');
    setDiscount('');
    setMinOrder('200000');
    setModalVisible(false);
  };

  const handleCreateVoucher = async () => {
    const cleanCode = code.trim().toUpperCase();
    if (!cleanCode) {
      showAlert('Lỗi', 'Vui lòng nhập mã voucher!');
      return;
    }
    const numDiscount = Number(discount);
    if (!numDiscount || numDiscount <= 0) {
      showAlert('Lỗi', 'Vui lòng nhập mức giảm giá hợp lệ!');
      return;
    }
    // Validate minOrder: không âm
    const numMinOrder = parseInt(minOrder, 10);
    if (isNaN(numMinOrder) || numMinOrder < 0) {
      showAlert('Lỗi', 'Đơn tối thiểu phải là số không âm!');
      return;
    }
    // Kiểm tra trùng mã voucher
    const isDuplicate = vouchers.some((v) => v.code.toUpperCase() === cleanCode);
    if (isDuplicate) {
      showAlert('Lỗi', `Mã voucher "${cleanCode}" đã tồn tại! Vui lòng dùng mã khác.`);
      return;
    }

    setIsSubmitting(true);
    const newVoucher: Voucher = {
      id: String(Date.now()),
      code: cleanCode,
      discount: numDiscount,
      status: 'active',
      minOrderAmount: numMinOrder,
    };

    try {
      await axiosClient.post('/vouchers', newVoucher);
      setVouchers((prev) => [...prev, newVoucher]);
      showAlert('Thành công', `Đã thêm voucher "${cleanCode}"!`);
    } catch (e: any) {
      console.warn('Lỗi gọi API thêm voucher, thêm vào state local:', e);
      setVouchers((prev) => [...prev, newVoucher]);
      showAlert('Thông báo', `Đã lưu voucher "${cleanCode}" vào bộ nhớ tạm (JSON Server offline)!`);
    } finally {
      setIsSubmitting(false);
      resetForm();
    }
  };

  const handleDeleteVoucher = (voucherId: string, voucherCode: string) => {
    const doDelete = async () => {
      try {
        await axiosClient.delete(`/vouchers/${voucherId}`);
        setVouchers((prev) => prev.filter((v) => v.id !== voucherId));
      } catch (e: any) {
        console.warn('Lỗi gọi API xóa voucher:', e);
        // Xóa khỏi local state để UI phản hồi mượt mà
        setVouchers((prev) => prev.filter((v) => v.id !== voucherId));
        showAlert('Thông báo', 'Đã xóa voucher khỏi bộ nhớ tạm (JSON Server offline)!');
      }
    };

    if (Platform.OS === 'web') {
      if (window.confirm(`Bạn có chắc muốn xóa mã giảm giá "${voucherCode}"?`)) {
        doDelete();
      }
    } else {
      Alert.alert('Xác nhận xóa', `Bạn có chắc muốn xóa voucher "${voucherCode}"?`, [
        { text: 'Hủy', style: 'cancel' },
        { text: 'Xóa', style: 'destructive', onPress: doDelete },
      ]);
    }
  };

  return (
    <SafeAreaView style={[styles.container, { backgroundColor: colors.bg }]}>
      <Header
        title="Voucher LumiShop"
        showBack={true}
        showCart={false}
        rightAction={
          <Pressable
            accessible={true}
            accessibilityRole="button"
            accessibilityLabel="Thêm voucher mới"
            onPress={() => setModalVisible(true)}
            style={[
              styles.topAddBtn,
              {
                backgroundColor: colors.accent,
                borderRadius: borderRadius.pill,
              },
            ]}
          >
            <Text style={styles.topAddText}>+ Tạo mã</Text>
          </Pressable>
        }
      />

      {isLoading ? (
        <View style={styles.centerBox}>
          <ActivityIndicator size="large" color={colors.accent} />
          <Text style={[styles.loadingText, { color: colors.muted }]}>
            Đang tải danh sách voucher...
          </Text>
        </View>
      ) : (
        <FlatList
          data={vouchers}
          keyExtractor={(item) => String(item.id)}
          contentContainerStyle={styles.listContent}
          renderItem={({ item }) => (
            <View
              style={[
                styles.voucherCard,
                {
                  backgroundColor: colors.card,
                  borderRadius: borderRadius.card,
                  borderColor: colors.border,
                },
                shadowSm,
              ]}
            >
              <View style={styles.voucherLeft}>
                <View
                  style={[
                    styles.codeBadge,
                    {
                      backgroundColor: colors.accentLight,
                      borderColor: colors.accent,
                      borderRadius: borderRadius.pill,
                    },
                  ]}
                >
                  <Text style={[styles.codeText, { color: colors.accent }]}>
                    {item.code}
                  </Text>
                </View>

                <View style={styles.voucherInfo}>
                  <Text style={[styles.discountText, { color: colors.text }]}>
                    Giảm {item.discount <= 100 ? `${item.discount}%` : formatCurrency(item.discount)}

                  <Text style={[styles.conditionText, { color: colors.muted }]}>
                    Đơn tối thiểu: {item.minOrderAmount ? formatCurrency(item.minOrderAmount) : '0đ'}
                  </Text>
                  <Text style={[styles.statusTag, { color: item.status === 'active' ? colors.success : colors.danger }]}>
                    ● {item.status === 'active' ? 'Đang hoạt động' : 'Hết hạn'}
                  </Text>
                </View>
              </View>

              <Pressable
                accessible={true}
                accessibilityRole="button"
                accessibilityLabel={`Xóa mã ${item.code}`}
                onPress={() => handleDeleteVoucher(item.id, item.code)}
                style={({ pressed }) => [
                  styles.deleteCircleBtn,
                  {
                    backgroundColor: '#FEE2E2',
                    transform: [{ scale: pressed ? 0.9 : 1 }],
                  },

              >
                <Text style={{ fontSize: 16 }}>🗑️</Text>
              </Pressable>
            </View>
          )}
        />
      )}

      {/* Modal Tạo Voucher Mới */}
      <Modal visible={modalVisible} animationType="slide" transparent>
        <View style={styles.modalOverlay}>
          <View
            style={[
              styles.modalCard,
              {
                backgroundColor: colors.card,
                borderRadius: borderRadius.card,
                borderColor: colors.border,
              },
              shadow,
            ]}
          >
            <View style={styles.modalHeader}>
              <Text style={[styles.modalTitle, { color: colors.text }]}>Tạo Voucher Mới</Text>
              <Pressable onPress={resetForm} style={styles.closeBtn}>
                <Text style={{ fontSize: 18, color: colors.muted }}>✕</Text>
              </Pressable>
            </View>

            <ScrollView contentContainerStyle={styles.formScroll}>
              <View style={styles.inputGroup}>
                <Text style={[styles.label, { color: colors.text }]}>Mã Voucher (In hoa) *</Text>
                <TextInput

// Màn hình Thanh toán LumiShop: Form giao hàng bo góc 12, Voucher có chip kiểm tra và Nút đặt hàng bo tròn 24

import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  ScrollView,
  StyleSheet,
  Pressable,
  ActivityIndicator,
  SafeAreaView,
  Alert,
  Platform,
} from 'react-native';
import { useRouter } from 'expo-router';
import { useCartStore } from '../../store/cartStore';
import { useAuthStore } from '../../store/authStore';
import { useTheme } from '../../theme/useTheme';
import { Header } from '../../components/Header';
import { useCheckVoucher } from '../../hooks/useCheckVoucher';
import { useCreateOrder } from '../../hooks/useCreateOrder';
import { calculateTotal, formatCurrency } from '../../utils/calculateTotal';
import { Voucher, Order } from '../../types';

export default function CheckoutScreen() {
  const router = useRouter();
  const { colors, borderRadius, shadow, shadowSm } = useTheme();
  const { items, clearCart } = useCartStore();
  const { user } = useAuthStore();

  const { checkVoucher, isLoading: isCheckingVoucher } = useCheckVoucher();
  const { createOrder, isLoading: isCreatingOrder } = useCreateOrder();

  // Form giao hàng
  const [customerName, setCustomerName] = useState(user?.name || user?.username || '');
  const [customerPhone, setCustomerPhone] = useState('0901234567');
  const [customerAddress, setCustomerAddress] = useState('Số 123 Đường Lê Lợi, Quận 1, TP. Hồ Chí Minh');

  // Trạng thái voucher
  const [voucherCodeInput, setVoucherCodeInput] = useState('');
  const [appliedVoucher, setAppliedVoucher] = useState<Voucher | null>(null);
  const [voucherMessage, setVoucherMessage] = useState<{ text: string; isError: boolean } | null>(null);

  // Tính tiền
  const { subtotal, discountAmount, total } = calculateTotal(items, appliedVoucher);

  // Xử lý áp dụng voucher
  const handleApplyVoucher = async () => {
    if (!voucherCodeInput.trim()) {
      setVoucherMessage({ text: 'Vui lòng nhập mã voucher!', isError: true });
      return;
    }

    try {
      setVoucherMessage(null);
      const voucher = await checkVoucher({
        code: voucherCodeInput.trim(),
        orderAmount: subtotal,
      });

      setAppliedVoucher(voucher);
      setVoucherMessage({
        text: `Đã áp dụng mã "${voucher.code}" thành công!`,
        isError: false,
      });
    } catch (err: any) {
      setAppliedVoucher(null);
      setVoucherMessage({
        text: err.message || 'Mã giảm giá không hợp lệ!',
        isError: true,
      });
    }
  };

  const handleRemoveVoucher = () => {
    setAppliedVoucher(null);
    setVoucherCodeInput('');
    setVoucherMessage(null);
  };

  // Xử lý đặt hàng
  const handlePlaceOrder = async () => {
    if (items.length === 0) {
      if (Platform.OS === 'web') {
        alert('Giỏ hàng trống: Vui lòng thêm sản phẩm vào giỏ trước khi thanh toán!');
      } else {
        Alert.alert('Giỏ hàng trống', 'Vui lòng thêm sản phẩm vào giỏ trước khi thanh toán!');
      }
      return;
    }

    if (!customerName.trim() || !customerPhone.trim() || !customerAddress.trim()) {
      if (Platform.OS === 'web') {
        alert('Thiếu thông tin: Vui lòng điền đầy đủ họ tên, số điện thoại và địa chỉ giao hàng!');
      } else {
        Alert.alert('Thiếu thông tin', 'Vui lòng điền đầy đủ họ tên, số điện thoại và địa chỉ giao hàng!');
      }
      return;
    }

    const newOrder: Order = {
      items: [...items],
      totalAmount: subtotal,
      discountAmount,
      finalAmount: total,
      customerName: customerName.trim(),
      customerPhone: customerPhone.trim(),
      customerAddress: customerAddress.trim(),
      voucherCode: appliedVoucher ? appliedVoucher.code : undefined,
      status: 'pending',
      createdAt: new Date().toISOString(),
    };

    try {
      const result = await createOrder(newOrder);
      clearCart();

      const successTitle = '🎉 Đặt hàng LumiShop thành công!';
      const successMsg = `Đơn hàng #${result.id} đã được tiếp nhận.\nTổng thanh toán: ${formatCurrency(total)}`;

      if (Platform.OS === 'web') {
        alert(`${successTitle}\n${successMsg}`);
        router.replace('/(app)');
      } else {
        Alert.alert(successTitle, successMsg, [
          {
            text: 'Về trang chủ LumiShop',
            onPress: () => router.replace('/(app)'),
          },
        ]);
      }
    } catch (err: any) {
      const errorMsg = err.message || 'Không thể tạo đơn hàng, vui lòng thử lại sau!';
      if (Platform.OS === 'web') {
        alert(`Lỗi đặt hàng: ${errorMsg}`);
      } else {
        Alert.alert('Lỗi đặt hàng', errorMsg);
      }
    }
  };

  if (items.length === 0) {
    return (
      <SafeAreaView style={[styles.container, { backgroundColor: colors.bg }]}>
        <Header title="Thanh toán" showBack={true} />
        <View style={styles.emptyContainer}>
          <Text style={{ fontSize: 44, marginBottom: 12 }}>🛍️</Text>
          <Text style={[styles.emptyTitle, { color: colors.text }]}>
            Không có sản phẩm để thanh toán
          </Text>
          <Pressable
            onPress={() => router.replace('/(app)')}
            style={[styles.backShopBtn, { backgroundColor: colors.accent, borderRadius: borderRadius.button }]}
          >
            <Text style={styles.backShopBtnText}>Quay lại mua sắm</Text>
          </Pressable>
        </View>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={[styles.container, { backgroundColor: colors.bg }]}>
      <Header title="Thanh toán đơn hàng" showBack={true} />

      <ScrollView
        contentContainerStyle={styles.scrollContent}
        keyboardShouldPersistTaps="handled"
      >
        {/* Khối 1: Form địa chỉ giao hàng */}
        <View
          style={[
            styles.card,
            {
              backgroundColor: colors.card,
              borderRadius: borderRadius.card,
              borderColor: colors.border,
            },
            shadowSm,
          ]}
        >
          <View style={styles.cardHeader}>
            <Text style={styles.cardIcon}>📍</Text>
            <Text style={[styles.cardTitle, { color: colors.text }]}>
              Thông tin nhận hàng
            </Text>
          </View>

          <View style={styles.inputGroup}>
            <Text style={[styles.label, { color: colors.text }]}>Họ và tên người nhận *</Text>
            <TextInput
              style={[
                styles.textInput,
                {
                  backgroundColor: colors.bg,
                  borderColor: colors.border,
                  color: colors.text,
                  borderRadius: borderRadius.input,
                },
              ]}
              placeholder="Ví dụ: Nguyễn Văn A"
              placeholderTextColor={colors.muted}
              value={customerName}
              onChangeText={setCustomerName}
            />
          </View>

          <View style={styles.inputGroup}>
            <Text style={[styles.label, { color: colors.text }]}>Số điện thoại liên hệ *</Text>
            <TextInput
              style={[
                styles.textInput,
                {
                  backgroundColor: colors.bg,
                  borderColor: colors.border,
                  color: colors.text,
                  borderRadius: borderRadius.input,
                },
              ]}
              placeholder="Ví dụ: 0901234567"
              placeholderTextColor={colors.muted}
              keyboardType="phone-pad"
              value={customerPhone}
              onChangeText={setCustomerPhone}
            />
          </View>

          <View style={styles.inputGroup}>
            <Text style={[styles.label, { color: colors.text }]}>Địa chỉ giao hàng chi tiết *</Text>
            <TextInput
              style={[
                styles.textInput,
                styles.textArea,
                {
                  backgroundColor: colors.bg,
                  borderColor: colors.border,
                  color: colors.text,
                  borderRadius: borderRadius.input,
                },
              ]}
              placeholder="Số nhà, tên đường, phường/xã, quận/huyện..."
              placeholderTextColor={colors.muted}
              multiline
              numberOfLines={3}
              value={customerAddress}
              onChangeText={setCustomerAddress}
            />
          </View>
        </View>

        {/* Khối 2: Ô nhập Voucher */}
        <View
          style={[
            styles.card,

              backgroundColor: colors.card,
              borderRadius: borderRadius.card,

              Mã giảm giá LumiShop
            </Text>
          </View>

          <View style={styles.voucherInputRow}>
            <TextInput
              style={[
                styles.voucherInput,
                {
                  backgroundColor: colors.bg,
                  borderColor: colors.border,
                  color: colors.text,
                  borderRadius: borderRadius.input,
                },
              ]}
              placeholder="Nhập mã (GIAM20, FREESHIP)..."
              placeholderTextColor={colors.muted}
              autoCapitalize="characters"
              value={voucherCodeInput}
              onChangeText={setVoucherCodeInput}
              editable={!appliedVoucher}
            />

            {appliedVoucher ? (
              <Pressable
                accessible={true}
                accessibilityRole="button"
                accessibilityLabel="Gỡ bỏ voucher"

                  styles.voucherBtn,

              </Pressable>
            ) : (
              <Pressable
                accessible={true}
                accessibilityRole="button"
                accessibilityLabel="Áp dụng mã"
                disabled={isCheckingVoucher}
                onPress={handleApplyVoucher}
                style={[
                  styles.voucherBtn,
                  {
                    backgroundColor: colors.accent,
                    borderRadius: borderRadius.input,
                    opacity: isCheckingVoucher ? 0.7 : 1,
                  },
                ]}
              >

                  <Text style={styles.voucherBtnText}>Áp dụng</Text>
                )}
              </Pressable>
            )}
          </View>

          {/* Chip hiển thị voucher đã áp dụng */}
          {appliedVoucher && (
            <View style={[styles.appliedChip, { backgroundColor: '#D1FAE5', borderColor: '#10B981', borderRadius: borderRadius.pill }]}>

              <Pressable onPress={handleRemoveVoucher} hitSlop={6}>

// app/_layout.tsx
// Root layout thuần React Native & TanStack Query (Không dùng UI library ngoài, kiểm soát 100% giao diện)

import React, { useEffect, useState } from 'react';
import { Stack } from 'expo-router';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { StatusBar } from 'expo-status-bar';
import { useAuthStore } from '../store/authStore';
import { useTheme } from '../theme/useTheme';

export default function RootLayout() {
  const { isDark, colors } = useTheme();
  const checkAuth = useAuthStore((state) => state.checkAuth);

  const [queryClient] = useState(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: {
            retry: 1,
            staleTime: 1000 * 60 * 3,
            refetchOnWindowFocus: false,
          },
        },
      })
  );

  useEffect(() => {
    checkAuth();
  }, [checkAuth]);

  return (
    <SafeAreaProvider>
      <QueryClientProvider client={queryClient}>
        <StatusBar style={isDark ? 'light' : 'dark'} />
        <Stack
          screenOptions={{
            headerShown: false,
            animation: 'slide_from_right',
            contentStyle: {
              backgroundColor: colors.bg,
            },
          }}
        >
          <Stack.Screen name="(auth)" options={{ headerShown: false }} />
          <Stack.Screen name="(app)" options={{ headerShown: false }} />
        </Stack>
      </QueryClientProvider>
    </SafeAreaProvider>
  );
}

import { useRouter } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';
import { useTheme } from '../theme/useTheme';
import { useCartStore } from '../store/cartStore';
import { useAuthStore } from '../store/authStore';

interface HeaderProps {
  title?: string;
  showBack?: boolean;
  showCart?: boolean;
  showAdmin?: boolean;
  showLogout?: boolean;
  rightAction?: React.ReactNode;
}

export const Header: React.FC<HeaderProps> = ({
  title,
  showBack = false,
  showCart = true,
  showAdmin = false,
  showLogout = false,
  rightAction,
}) => {
  const router = useRouter();
  const { colors, shadowSm } = useTheme();
  const totalItems = useCartStore((state) => state.getTotalItems());
  const { user, logout } = useAuthStore();

  const handleLogout = async () => {
    if (Platform.OS === 'web') {
      if (window.confirm('Bạn có chắc chắn muốn đăng xuất khỏi LumiShop?')) {
        await logout();
        router.replace('/(auth)/sign-in');
      }
    } else {
      Alert.alert('Đăng xuất', 'Bạn có chắc chắn muốn đăng xuất khỏi LumiShop?', [
        { text: 'Hủy', style: 'cancel' },
        {
          text: 'Đăng xuất',
          style: 'destructive',
          onPress: async () => {
            await logout();
            router.replace('/(auth)/sign-in');
          },
        },
      ]);
    }
  };

  return (
    <View
      style={[
        styles.container,
        {
          backgroundColor: colors.card,
          borderBottomColor: colors.border,
        },
        shadowSm,
      ]}
    >
      {/* Bên trái: Nút Back hoặc Logo LumiShop nằm sát bên trái */}
      <View style={styles.leftBox}>
        {showBack ? (
          <View style={styles.backGroup}>
            <Pressable
              accessible={true}
              accessibilityRole="button"
              accessibilityLabel="Quay lại"
              onPress={() => router.back()}
              hitSlop={8}
              style={({ pressed }) => [
                styles.iconBtn,
                {
                  backgroundColor: colors.bg,
                  transform: [{ scale: pressed ? 0.92 : 1 }],
                },
              ]}
            >
              <Ionicons name="arrow-back" size={22} color={colors.text} />
            </Pressable>
            {title ? (
              <Text numberOfLines={1} style={[styles.screenTitle, { color: colors.text }]}>
                {title}
              </Text>
            ) : null}
          </View>
        ) : (
          <Pressable onPress={() => router.push('/(app)')}>
            <Text style={[styles.brandLogo, { color: colors.accent }]}>
              LumiShop
            </Text>
          </Pressable>
        )}
      </View>

      {/* Ở giữa: Tiêu đề màn hình (nếu có khi không ở chế độ back) */}
      {!showBack && title ? (
        <View style={styles.centerBox}>
          <Text numberOfLines={1} style={[styles.screenTitle, { color: colors.text }]}>
            {title}
          </Text>
        </View>
      ) : null}

      {/* Bên phải: 3 icon Ionicons nằm ngang, gap: 16 sát mép phải */}
      <View style={styles.rightBox}>
        {rightAction ? (
          rightAction
        ) : (
          <View style={styles.iconRow}>
            {/* Icon Admin */}
            {(showAdmin || user?.role === 'admin') && !showBack && (
              <Pressable
                accessible={true}
                accessibilityRole="button"
                accessibilityLabel="Quản trị hệ thống"
                onPress={() => router.push('/(app)/admin')}
                hitSlop={8}
                style={({ pressed }) => [
                  styles.iconBtn,
                  {
                    backgroundColor: colors.bg,
                    transform: [{ scale: pressed ? 0.92 : 1 }],
                  },
                ]}
              >
                <Ionicons name="settings-outline" size={22} color={colors.text} />
              </Pressable>
            )}

            {/* Icon Giỏ hàng kèm Badge số lượng màu đỏ */}
            {showCart && (
              <Pressable
                accessible={true}
                accessibilityRole="button"
                accessibilityLabel={`Giỏ hàng ${totalItems} sản phẩm`}
                onPress={() => router.push('/(app)/cart')}
                hitSlop={8}
                style={({ pressed }) => [
                  styles.iconBtn,
                  {
                    backgroundColor: colors.bg,
                    transform: [{ scale: pressed ? 0.92 : 1 }],
                  },
                ]}
              >
                <Ionicons name="cart-outline" size={24} color={colors.text} />
                {totalItems > 0 && (
                  <View style={[styles.badge, { backgroundColor: colors.danger }]}>
                    <Text style={styles.badgeText}>
                      {totalItems > 99 ? '99+' : totalItems}
                    </Text>
                  </View>
                )}
              </Pressable>
            )}

            {/* Icon Đăng xuất */}
            {(showLogout || !showBack) && (
              <Pressable
                accessible={true}
                accessibilityRole="button"
                accessibilityLabel="Đăng xuất tài khoản"
                onPress={handleLogout}
                hitSlop={8}
                style={({ pressed }) => [
                  styles.iconBtn,
                  {
                    backgroundColor: colors.bg,
                    transform: [{ scale: pressed ? 0.92 : 1 }],
                  },
                ]}
              >
                <Ionicons name="log-out-outline" size={22} color={colors.text} />
              </Pressable>
            )}
          </View>
        )}
      </View>
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    height: 60,

// components/ProtectedRoute.tsx
// Component bọc bảo vệ route, dùng StyleSheet tùy chỉnh và ActivityIndicator React Native thuần

import React, { useEffect } from 'react';
import { View, ActivityIndicator, Text, StyleSheet } from 'react-native';
import { useRouter } from 'expo-router';
import { useAuthStore } from '../store/authStore';
import { useTheme } from '../theme/useTheme';

interface ProtectedRouteProps {
  children: React.ReactNode;
}

export const ProtectedRoute: React.FC<ProtectedRouteProps> = ({ children }) => {
  const router = useRouter();
  const { colors } = useTheme();
  const { token, isLoading } = useAuthStore();

  useEffect(() => {
    if (!isLoading && !token) {
      router.replace('/(auth)/sign-in');
    }
  }, [isLoading, token, router]);

  if (isLoading) {
    return (
      <View style={[styles.loadingContainer, { backgroundColor: colors.bg }]}>
        <ActivityIndicator size="large" color={colors.accent} />
        <Text style={[styles.loadingText, { color: colors.muted }]}>
          Đang xác thực tài khoản LumiShop...
        </Text>
      </View>
    );
  }

  if (!token) {
    return null;
  }

  return <>{children}</>;
};

const styles = StyleSheet.create({
  loadingContainer: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
    padding: 24,
  },
  loadingText: {
    marginTop: 14,
    fontSize: 14,
    fontWeight: '500',
  },
});

export default ProtectedRoute;

// hooks/useCheckVoucher.ts
// Hook kiểm tra tính hợp lệ của mã giảm giá (Voucher) bằng useMutation

import { useMutation } from '@tanstack/react-query';
import axiosClient from '../api/axiosClient';
import { Voucher } from '../types';
import { initialVouchers } from '../data/products';

interface CheckVoucherPayload {
  code: string;
  orderAmount?: number;
}

async function checkVoucherApi({ code, orderAmount = 0 }: CheckVoucherPayload): Promise<Voucher> {
  const cleanCode = code.trim().toUpperCase();
  if (!cleanCode) {
    throw new Error('Vui lòng nhập mã giảm giá!');
  }

  let voucher: Voucher | null = null;

  try {
    const response = await axiosClient.get<Voucher[]>(`/vouchers?code=${encodeURIComponent(cleanCode)}`);
    if (response.data && response.data.length > 0) {
      voucher = response.data[0];
    }
  } catch (error) {
    console.warn('Lỗi gọi API voucher, thử kiểm tra trong danh sách voucher dự phòng:', error);
  }

  // Fallback từ initialVouchers nếu không gọi được API
  if (!voucher) {
    const fallback = initialVouchers.find((v) => v.code.toUpperCase() === cleanCode);
    if (fallback) {
      voucher = fallback;
    }
  }

  if (!voucher) {
    throw new Error(`Mã giảm giá "${cleanCode}" không tồn tại hoặc đã hết hạn!`);
  }

  if (voucher.status !== 'active') {
    throw new Error(`Mã giảm giá "${cleanCode}" hiện không khả dụng!`);
  }

  if (voucher.minOrderAmount && orderAmount < voucher.minOrderAmount) {
    throw new Error(
      `Mã này chỉ áp dụng cho đơn hàng từ ${new Intl.NumberFormat('vi-VN').format(voucher.minOrderAmount)}đ trở lên!`
    );
  }

  return voucher;
}

export function useCheckVoucher() {
  const mutation = useMutation<Voucher, Error, CheckVoucherPayload>({
    mutationFn: checkVoucherApi,
  });

  return {
    checkVoucher: mutation.mutateAsync,
    voucher: mutation.data,
    isLoading: mutation.isPending,
    isError: mutation.isError,
    error: mutation.error,
    reset: mutation.reset,
  };
}

export default useCheckVoucher;

import { useMutation, useQueryClient } from "@tanstack/react-query";
import axiosClient from "../api/axiosClient";
import { Order } from "../types";

async function createOrderApi(orderData: Order): Promise<Order> {
  try {
    const response = await axiosClient.post<Order>("/orders", {
      ...orderData,
      id: orderData.id || String(Date.now()),
      createdAt: orderData.createdAt || new Date().toISOString(),
    });

    // Trừ tồn kho cho từng sản phẩm trong đơn hàng (json-server không tự làm)
    if (orderData.items && orderData.items.length > 0) {
      await Promise.allSettled(
        orderData.items.map(async (item) => {
          try {
            const productRes = await axiosClient.get(
              `/products/${item.product.id}`,
            );
            const currentStock = productRes.data?.stock ?? 0;
            const newStock = Math.max(0, currentStock - item.quantity);
            await axiosClient.patch(`/products/${item.product.id}`, {
              stock: newStock,
            });
          } catch (stockErr) {
            console.warn(
              `Không trừ được kho cho sản phẩm ${item.product.id}:`,
              stockErr,
            );
          }
        }),
      );
    }

    return response.data;
  } catch (error) {
    console.warn(
      "Lỗi khi gọi API POST /orders, lưu tạm đơn hàng offline:",
      error,
    );
    // Giả lập đơn hàng thành công khi JSON Server offline
    return {
      ...orderData,
      id: orderData.id || `ORDER-${Date.now()}`,
      createdAt: new Date().toISOString(),
      status: "pending",
    };
  }
}

export function useCreateOrder() {
  const queryClient = useQueryClient();

  const mutation = useMutation<Order, Error, Order>({
    mutationFn: createOrderApi,
    onSuccess: () => {
      // Làm mới dữ liệu cache sản phẩm sau khi đặt hàng thành công
      queryClient.invalidateQueries({ queryKey: ["products"] });
      queryClient.invalidateQueries({ queryKey: ["product"] });
    },
  });

  return {
    createOrder: mutation.mutateAsync,
    order: mutation.data,
    isLoading: mutation.isPending,
    isSuccess: mutation.isSuccess,
    isError: mutation.isError,
    error: mutation.error,
    reset: mutation.reset,
  };
}

export default useCreateOrder;

import axiosClient from '../api/axiosClient';

import { initialProducts } from '../data/products';

async function fetchProductById(id: string): Promise<Product> {
  try {
    const response = await axiosClient.get<Product>(`/products/${id}`);
    if (response.data) {
      return response.data;
    }
  } catch (error) {
    console.warn(`Lỗi fetch sản phẩm ID ${id} từ server, fallback seed:`, error);
  }

  // Fallback từ seed data nếu API lỗi hoặc không tìm thấy
  const fallback = initialProducts.find((p) => String(p.id) === String(id));
  if (fallback) {
    return fallback;
  }

  throw new Error(`Không tìm thấy sản phẩm có mã #${id}`);
}

export function useProduct(id: string) {
  const query = useQuery<Product, Error>({
    queryKey: ['product', id],
    queryFn: () => fetchProductById(id),
    enabled: !!id,
    staleTime: 1000 * 60 * 5,
  });

  return {
    product: query.data,
    isLoading: query.isLoading,
    isError: query.isError,
    error: query.error,
    refetch: query.refetch,
  };
}

import { useQuery } from '@tanstack/react-query';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { Platform } from 'react-native';
import axiosClient from '../api/axiosClient';
import { Product } from '../types';
import { initialProducts } from '../data/products';

// Khóa lưu trữ cache sản phẩm theo yêu cầu đề bài
export const PRODUCTS_CACHE_KEY = 'lostfound-shopabc-products';

// Helper an toàn đọc cache từ Storage (hỗ trợ cả Web & Mobile)
async function getCachedProducts(): Promise<Product[] | null> {
  try {
    const raw =
      Platform.OS === 'web' && typeof window !== 'undefined' && window.localStorage
        ? window.localStorage.getItem(PRODUCTS_CACHE_KEY)
        : await AsyncStorage.getItem(PRODUCTS_CACHE_KEY);

    if (raw) {
      const parsed = JSON.parse(raw);
      if (Array.isArray(parsed) && parsed.length > 0) {
        return parsed as Product[];
      }
    }
  } catch (e) {
    console.warn('Lỗi đọc cache sản phẩm từ storage:', e);
  }
  return null;
}

// Helper an toàn lưu cache vào Storage
async function setCachedProducts(products: Product[]): Promise<void> {
  try {
    const raw = JSON.stringify(products);
    if (Platform.OS === 'web' && typeof window !== 'undefined' && window.localStorage) {
      window.localStorage.setItem(PRODUCTS_CACHE_KEY, raw);
    } else {
      await AsyncStorage.setItem(PRODUCTS_CACHE_KEY, raw);
    }
  } catch (error) {
    console.warn('Lỗi khi ghi cache sản phẩm:', error);
  }
}

// Hàm fetch danh sách sản phẩm: đọc từ API, nếu offline tự khôi phục từ AsyncStorage hoặc initialProducts
async function fetchProducts(): Promise<Product[]> {
  try {
    const response = await axiosClient.get<Product[]>('/products');
    if (response.data && Array.isArray(response.data)) {
      await setCachedProducts(response.data);
      return response.data;
    }
    return initialProducts;
  } catch (error) {
    console.warn('JSON Server offline, đọc từ AsyncStorage cache hoặc seed ban đầu:', error);
    const cached = await getCachedProducts();
    if (cached && cached.length > 0) {
      return cached;
    }
    return initialProducts;
  }
}

export function useProducts() {
  const query = useQuery<Product[], Error>({
    queryKey: ['products'],
    queryFn: fetchProducts,
    staleTime: 1000 * 60 * 5, // 5 phút cache tươi
  });

  return {
    data: query.data ?? initialProducts,
    isLoading: query.isLoading,
    isError: query.isError,
    error: query.error,
    refetch: query.refetch,
    isFetching: query.isFetching,
  };
}

export default useProducts;

import { create } from 'zustand';

const USER_KEY = 'shopabc_auth_user';

// Helper: Lưu token an toàn (Mobile dùng SecureStore, Web dùng localStorage)
export async function saveToken(token: string): Promise<void> {
  try {
    if (Platform.OS === 'web') {
      if (typeof window !== 'undefined' && window.localStorage) {
        window.localStorage.setItem(TOKEN_KEY, token);
      }
    } else {
      await SecureStore.setItemAsync(TOKEN_KEY, token);
    }
  } catch (error) {
    console.error('Lỗi khi lưu token:', error);
  }
}

// Helper: Đọc token
export async function getToken(): Promise<string | null> {
  try {
    if (Platform.OS === 'web') {
      if (typeof window !== 'undefined' && window.localStorage) {
        return window.localStorage.getItem(TOKEN_KEY);
      }
      return null;
    } else {
      return await SecureStore.getItemAsync(TOKEN_KEY);
    }
  } catch (error) {
    console.error('Lỗi khi đọc token:', error);
    return null;
  }
}

// Helper: Xóa token khi đăng xuất
export async function deleteToken(): Promise<void> {
  try {
    if (Platform.OS === 'web') {
      if (typeof window !== 'undefined' && window.localStorage) {
        window.localStorage.removeItem(TOKEN_KEY);
      }
    } else {
      await SecureStore.deleteItemAsync(TOKEN_KEY);
    }
  } catch (error) {
    console.error('Lỗi khi xóa token:', error);
  }
}

  getTotalItems: () => number;
  getTotalAmount: () => number;
}

// Cấu hình storage tương thích tốt cả Web (localStorage) và Mobile (AsyncStorage)
const customStorage = {
  getItem: async (name: string): Promise<string | null> => {
    if (Platform.OS === 'web') {
      try {
        if (typeof window !== 'undefined' && window.localStorage) {
          return window.localStorage.getItem(name);
        }
      } catch (e) {
        console.warn('Lỗi khi đọc localStorage trên Web:', e);
      }
    }
    return AsyncStorage.getItem(name);
  },
  setItem: async (name: string, value: string): Promise<void> => {
    if (Platform.OS === 'web') {
      try {
        if (typeof window !== 'undefined' && window.localStorage) {
          window.localStorage.setItem(name, value);
          return;
        }
      } catch (e) {
        console.warn('Lỗi khi ghi localStorage trên Web:', e);
      }
    }
    return AsyncStorage.setItem(name, value);
  },
  removeItem: async (name: string): Promise<void> => {
    if (Platform.OS === 'web') {
      try {
        if (typeof window !== 'undefined' && window.localStorage) {
          window.localStorage.removeItem(name);
          return;
        }
      } catch (e) {
        console.warn('Lỗi khi xóa localStorage trên Web:', e);
      }
    }

  },

export const useCartStore = create<CartState>()(
  persist(
    (set, get) => ({
      items: [],

      // Thêm sản phẩm vào giỏ hàng
      addItem: (product: Product, quantity: number = 1) => {
        // Chặn thêm hàng hết kho
        if ((product.stock ?? 1) <= 0) {
          return;
        }
        const currentItems = get().items;
        const existingIndex = currentItems.findIndex(
          (item) => item.product.id === product.id
        );

        if (existingIndex > -1) {
          const updatedItems = [...currentItems];
          const newQuantity = updatedItems[existingIndex].quantity + quantity;
          // Không vượt quá số lượng tồn kho nếu có giới hạn
          const maxStock = product.stock ?? 999;
          updatedItems[existingIndex] = {
            ...updatedItems[existingIndex],
            product, // Cập nhật thông tin product mới nhất (stock, price, ...)
            quantity: Math.min(newQuantity, maxStock),
          };
          set({ items: updatedItems });
        } else {
          const newItem: CartItem = {
            id: `${product.id}-${Date.now()}`,
            product,
            quantity: Math.min(quantity, product.stock ?? 999),
          };
          set({ items: [...currentItems, newItem] });
        }
      },

      // Xóa sản phẩm khỏi giỏ theo id sản phẩm
      removeItem: (productId: string) => {
        const updatedItems = get().items.filter(
          (item) => item.product.id !== productId
        );
        set({ items: updatedItems });
      },

      // Cập nhật số lượng của một sản phẩm
      updateQuantity: (productId: string, quantity: number) => {
        if (quantity <= 0) {
          get().removeItem(productId);
          return;
        }

        const currentItems = get().items;
        const updatedItems = currentItems.map((item) => {
          if (item.product.id === productId) {
            const maxStock = item.product.stock ?? 999;
            return {
              ...item,
              quantity: Math.min(quantity, maxStock),
            };
          }
          return item;
        });
        set({ items: updatedItems });
      },

      // Xóa sạch giỏ hàng
      clearCart: () => {
        set({ items: [] });
      },

      // Đếm tổng số lượng món đồ trong giỏ
      getTotalItems: () => {
        return get().items.reduce((total, item) => total + item.quantity, 0);
      },

      // Tính tổng tiền toàn bộ giỏ hàng
      getTotalAmount: () => {
        return get().items.reduce(
          (total, item) => total + item.product.price * item.quantity,
          0
        );
      },
    }),
    {
      name: 'shopabc-cart-storage',
      storage: createJSONStorage(() => customStorage),
    }
  )
);

export default useCartStore;

// Hook cung cấp bảng màu, bán kính bo góc và hiệu ứng đổ bóng mềm cho LumiShop

import { useColorScheme, Platform, ViewStyle } from 'react-native';
import { colors, ColorTheme } from './colors';

export interface UseThemeReturn {
  colors: ColorTheme;
  isDark: boolean;
  colorScheme: 'light' | 'dark';
  borderRadius: {
    input: number;
    card: number;
    button: number;
    pill: number;
    full: number;
  };
  shadow: ViewStyle;
  shadowSm: ViewStyle;
}

export function useTheme(): UseThemeReturn {
  const scheme = useColorScheme();
  const isDark = scheme === 'dark';
  const activeColors = isDark ? colors.dark : colors.light;

  const shadow: ViewStyle = Platform.select({
    ios: {
      shadowColor: '#000',
      shadowOffset: { width: 0, height: 4 },
      shadowOpacity: isDark ? 0.3 : 0.05,
      shadowRadius: 12,
    },
    android: {
      elevation: 3,
    },
    web: {
      boxShadow: isDark
        ? '0 4px 14px rgba(0, 0, 0, 0.4)'
        : '0 4px 16px rgba(0, 0, 0, 0.05)',
    } as any,
    default: {},
  });

  const shadowSm: ViewStyle = Platform.select({
    ios: {
      shadowColor: '#000',
      shadowOffset: { width: 0, height: 2 },
      shadowOpacity: isDark ? 0.2 : 0.04,
      shadowRadius: 6,
    },
    android: {
      elevation: 2,
    },
    web: {
      boxShadow: isDark
        ? '0 2px 8px rgba(0, 0, 0, 0.3)'
        : '0 2px 8px rgba(0, 0, 0, 0.04)',
    } as any,
    default: {},
  });

  return {
    colors: activeColors,
    isDark,
    colorScheme: isDark ? 'dark' : 'light',
    borderRadius: {
      input: 12,
      card: 16,
      button: 24,
      pill: 20,
      full: 9999,
    },
    shadow,
    shadowSm,
  };
}

export default useTheme;

// utils/calculateTotal.ts
// Tiện ích tính toán tổng tiền, tiền giảm giá và tổng thanh toán của đơn hàng

import { CartItem, Voucher } from '../types';

export interface OrderCalculation {
  subtotal: number;       // Tổng tiền hàng tạm tính
  discountAmount: number; // Số tiền được giảm
  total: number;          // Tiền phải trả sau khi giảm
}

/**
 * Tính tổng tiền hàng tạm tính
 */
export function calculateSubtotal(items: CartItem[]): number {
  return items.reduce((sum, item) => sum + item.product.price * item.quantity, 0);
}

/**
 * Tính số tiền giảm giá theo voucher
 * @param subtotal Tổng tiền trước giảm
 * @param voucher Thông tin voucher (nếu có)
 */
export function calculateDiscount(subtotal: number, voucher?: Voucher | null): number {
  if (!voucher || voucher.status !== 'active') {
    return 0;
  }

  // Nếu voucher có yêu cầu đơn hàng tối thiểu
  if (voucher.minOrderAmount && subtotal < voucher.minOrderAmount) {
    return 0;
  }

  // Nếu discount <= 100 coi như tỷ lệ phần trăm (%), ngược lại coi như số tiền cụ thể
  if (voucher.discount <= 100) {
    return Math.round((subtotal * voucher.discount) / 100);
  } else {
    return Math.min(voucher.discount, subtotal);
  }
}

/**
 * Tính toàn bộ chi phí đơn hàng
 */
export function calculateTotal(items: CartItem[], voucher?: Voucher | null): OrderCalculation {
  const subtotal = calculateSubtotal(items);
  const discountAmount = calculateDiscount(subtotal, voucher);
  const total = Math.max(0, subtotal - discountAmount);

  return {
    subtotal,
    discountAmount,
    total,
  };
}

/**
 * Định dạng tiền tệ VND (Ví dụ: 150.000 đ)
 */
export function formatCurrency(amount: number): string {
  return new Intl.NumberFormat('vi-VN', {
    style: 'currency',
    currency: 'VND',
  }).format(amount);
}

export default calculateTotal;
